```
python - <<'PY'
import ast, pathlib
hits = set()
for p in pathlib.Path("butler").rglob("*.py"):
    try:
        t = ast.parse(p.read_text(encoding="utf-8", errors="ignore"))
    except Exception:
        continue
    for fn in ast.walk(t):
        if not isinstance(fn, (ast.FunctionDef, ast.AsyncFunctionDef)):
            continue
        for node in ast.walk(fn):
            if not isinstance(node, ast.Try):
                continue
            has_commit = any(
                isinstance(n, ast.Call) and isinstance(n.func, ast.Attribute)
                and n.func.attr == "commit" for n in ast.walk(node))
            if not has_commit or not node.handlers:
                continue
            if not any("rollback" in ast.dump(h) for h in node.handlers):
                hits.add((str(p), fn.lineno, fn.name))
print("sites:", len(hits))
for h in sorted(hits):
    print(h)
PY
```
